Salvar em qualquer dispositivo

Um arquivo HTML sozinho não sincroniza dados entre computadores/celulares. Esta versão usa Supabase como banco de dados na nuvem, com login. Você configura uma vez e depois acessa o mesmo projeto em qualquer dispositivo.

  1. Crie um projeto gratuito em supabase.com.
  2. No painel do projeto, abra SQL Editor, cole o SQL abaixo e execute.
  3. Em Project Settings → API, copie a Project URL e a chave publishable/anon (nunca use a service_role).
  4. Preencha os dois campos abaixo e clique em Salvar configuração. Depois crie sua conta por e-mail e senha.
create table if not exists public.analises_acoes (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade default auth.uid(),
  empresa text not null,
  ticker text not null,
  setor text,
  data_analise date not null default current_date,
  preco_mercado numeric,
  quantitativo numeric,
  max_quantitativo numeric default 60,
  qualitativo numeric,
  max_qualitativo numeric default 40,
  numero_graham numeric,
  graham_growth numeric,
  dcf numeric,
  multiplios_comparaveis numeric,
  margem_seguranca numeric generated always as (
    case when numero_graham is not null and numero_graham > 0 and preco_mercado is not null
      then (numero_graham - preco_mercado) / numero_graham
      else null end
  ) stored,
  risco_principal text,
  notas text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
alter table public.analises_acoes enable row level security;
create policy "Usuário vê apenas as próprias análises"
on public.analises_acoes for select to authenticated using (auth.uid() = user_id);
create policy "Usuário cria apenas as próprias análises"
on public.analises_acoes for insert to authenticated with check (auth.uid() = user_id);
create policy "Usuário atualiza apenas as próprias análises"
on public.analises_acoes for update to authenticated using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "Usuário exclui apenas as próprias análises"
on public.analises_acoes for delete to authenticated using (auth.uid() = user_id);
grant select, insert, update, delete on public.analises_acoes to authenticated;
